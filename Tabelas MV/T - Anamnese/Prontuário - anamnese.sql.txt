Query para consulta

SELECT 
  a.cd_paciente,
  a.cd_atendimento,
  A.hr_atendimento,
  PDC.CD_DOCUMENTO_CLINICO,
  PDC.cd_tipo_documento,
  PDC.cd_objeto,
  EC.CD_CAMPO,
  EC.CD_METADADO,
  ERC.lo_valor 
FROM DBAMV.ATENDIME A
INNER JOIN DBAMV.PW_DOCUMENTO_CLINICO PDC ON PDC.CD_ATENDIMENTO = A.CD_ATENDIMENTO
INNER JOIN DBAMV.PW_EDITOR_CLINICO PEC ON PEC.CD_DOCUMENTO_CLINICO = PDC.CD_DOCUMENTO_CLINICO
INNER JOIN DBAMV.EDITOR_REGISTRO_CAMPO ERC ON ERC.CD_REGISTRO = PEC.CD_EDITOR_REGISTRO
INNER JOIN DBAMV.EDITOR_CAMPO EC ON EC.CD_CAMPO = ERC.CD_CAMPO
WHERE pdc.cd_documento_clinico IN (4722459, 4984282, 5229532, 5356605, 5509355, 5789656, 5989748)    
-- PDC.CD_DOCUMENTO_CLINICO  = 5509355 
--AND A.cd_atendimento = 662810
--AND PDC.cd_tipo_documento      =  '30' 
--AND PDC.cd_objeto = 466 
--AND EC.CD_CAMPO IN (33190, 33206) 
AND EC.CD_METADADO IN (4889, 4921)
ORDER BY cd_atendimento asc;
